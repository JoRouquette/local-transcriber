# Moteur de transcription/diarisation (gele ensuite avec PyInstaller).
#
# STACK EPINGLEE ET TESTEE (transcription FR validee de bout en bout) sur Python 3.11.
# Ne pas utiliser Python 3.13 (whisperx/pyannote de pointe + torchcodec + DLL FFmpeg).
# FFmpeg (executable) doit etre sur le PATH.
#
# Notes de compatibilite (durement acquises) :
#   - whisperx 3.3.1 + faster-whisper 1.1.0 : signatures TranscriptionOptions alignees.
#   - av 12 (faster-whisper 1.1) a un wheel Windows cp311 (av 11 n'en a pas -> compile MSVC).
#   - torch 2.2.2 : evite torchcodec (tire par torchaudio >= 2.8).
#   - nltk 3.8.1 : evite le finder "inisec" (nltk 3.9+) qui casse import whisperx.
#   - speechbrain 1.1.0 (via pyannote 3.3) : ses lazy modules (k2_fsa, nlp) cassent
#     inspect.stack() ; neutralise au runtime par transcriber_engine._compat.
#
# PyTorch s'installe en amont selon la cible (voir README / build-engine.ps1) :
#   CPU      :  pip install torch==2.2.2 torchaudio==2.2.2 --index-url https://download.pytorch.org/whl/cpu
#   GPU CUDA :  pip install torch==2.2.2 torchaudio==2.2.2 --index-url https://download.pytorch.org/whl/cu121

whisperx==3.3.1
pyannote.audio==3.3.2

python-dotenv>=1.0.1

# setuptools < 80 : fournit `pkg_resources`, dont whisperx a besoin au runtime.
# `uv venv` n'installe PAS setuptools (contrairement a `python -m venv`), et
# setuptools >= 81 a SUPPRIME pkg_resources -> a epingler explicitement ici, sinon
# la transcription echoue avec "No module named 'pkg_resources'".
setuptools<80
wheel
